} LIMIT 50

- "Quali ingredienti della famiglia della frutta a guscio ('nut') o dei semi ('seed') mi consigli per avere tante proteine? Dimmi calorie, proteine e grassi."
 
SELECT DISTINCT ?nome ?calorie ?proteine ?grassi WHERE {
    ?ingrediente <http://www.w3.org/2000/01/rdf-schema#label> ?nome .
    ?ingrediente <http://www.w3.org/1999/02/22-rdf-syntax-ns#type> ?foodon .
    ?foodon <http://www.w3.org/2000/01/rdf-schema#subClassOf>* ?genitore .
    ?genitore <http://www.w3.org/2000/01/rdf-schema#label> ?famiglia .
    FILTER(CONTAINS(LCASE(STR(?famiglia)), "nut") || CONTAINS(LCASE(STR(?famiglia)), "seed") || CONTAINS(LCASE(STR(?nome)), "nut") || CONTAINS(LCASE(STR(?nome)), "seed"))
    FILTER(!CONTAINS(LCASE(STR(?nome)), "mix"))
    ?ingrediente <http://tesi.unibs.it/ontology#hasNutritionProfile> ?usda .
    OPTIONAL { ?usda <http://tesi.unibs.it/ontology#hasCalories> ?calorie . }
    OPTIONAL { ?usda <http://tesi.unibs.it/ontology#hasProtein> ?proteine . }
    OPTIONAL { ?usda <http://tesi.unibs.it/ontology#hasFat> ?grassi . }
}
ORDER BY DESC(?proteine)
LIMIT 50
